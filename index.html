// Google Apps Script untuk menyimpan riwayat praktik siswa ke Google Sheets. // 1) Ganti dua nilai di bawah. 2) Deploy sebagai Web App (Execute as: Me, Who has access: Anyone). const KODE_KELAS = 'GANTI_KODE_KELAS'; // dibagikan ke siswa const KUNCI_GURU = 'GANTI_KUNCI_GURU'; // rahasia, hanya untuk guru const HEAD_T = ['key','sid','nama','kelas','tiket','status','updated','json']; const HEAD_N = ['sid','nilai','komentar','updated']; function sh_(n, h) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let s = ss.getSheetByName(n); if (!s) { s = ss.insertSheet(n); s.appendRow(h); } return s; } function out_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); } function upsert_(s, key, row) { const f = s.createTextFinder(key).matchEntireCell(true).findNext(); if (f && f.getColumn() === 1) s.getRange(f.getRow(), 1, 1, row.length).setValues([row]); else s.appendRow(row); } function doPost(e) { const b = JSON.parse(e.postData.contents); const lock = LockService.getScriptLock(); lock.waitLock(20000); try { if (b.action === 'save') { if (b.kode !== KODE_KELAS) return out_({ ok: false, err: 'kode' }); const s = sh_('Tiket', HEAD_T); Object.keys(b.tickets || {}).forEach(id => { const t = b.tickets[id], key = b.sid + '|' + id; upsert_(s, key, [key, b.sid, b.nama, b.kelas, id, t.st, new Date(), JSON.stringify(t)]); }); return out_({ ok: true }); } if (b.action === 'grade') { if (b.key !== KUNCI_GURU) return out_({ ok: false, err: 'key' }); upsert_(sh_('Nilai', HEAD_N), b.sid, [b.sid, b.score, b.comment, new Date()]); return out_({ ok: true }); } return out_({ ok: false, err: 'action' }); } finally { lock.releaseLock(); } } function doGet(e) { const p = e.parameter; const rows = sh_('Tiket', HEAD_T).getDataRange().getValues().slice(1); const grows = sh_('Nilai', HEAD_N).getDataRange().getValues().slice(1); if (p.action === 'load') { if (p.kode !== KODE_KELAS) return out_({ ok: false, err: 'kode' }); const tickets = {}; let grade = null; rows.forEach(r => { if (r[1] === p.sid) tickets[r[4]] = JSON.parse(r[7]); }); grows.forEach(r => { if (r[0] === p.sid) grade = { score: r[1], comment: r[2] }; }); return out_({ ok: true, tickets: tickets, grade: grade }); } if (p.action === 'list') { if (p.key !== KUNCI_GURU) return out_({ ok: false, err: 'key' }); const m = {}, grades = {}; rows.forEach(r => { const x = m[r[1]] = m[r[1]] || { id: r[1], name: r[2], kelas: r[3], updated: 0, tickets: {} }; x.tickets[r[4]] = JSON.parse(r[7]); x.updated = Math.max(x.updated, new Date(r[6]).getTime()); }); grows.forEach(r => { grades[r[0]] = { score: r[1], comment: r[2] }; }); return out_({ ok: true, sessions: Object.values(m), grades: grades }); } return out_({ ok: false, err: 'action' }); }
